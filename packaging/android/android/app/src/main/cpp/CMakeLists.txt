cmake_minimum_required(VERSION 3.22.1)
project(folia_node)
set(NODE_RUNTIME "${CMAKE_CURRENT_LIST_DIR}/../../../../../.cache/runtime")
add_library(node SHARED IMPORTED)
set_target_properties(node PROPERTIES IMPORTED_LOCATION "${NODE_RUNTIME}/bin/${ANDROID_ABI}/libnode.so")
add_library(folia-node SHARED nodeRuntime.cpp)
target_include_directories(folia-node PRIVATE "${NODE_RUNTIME}/include/node")
target_compile_features(folia-node PRIVATE cxx_std_20)
target_link_libraries(folia-node node android log)
target_link_options(folia-node PRIVATE "-Wl,-z,max-page-size=16384")
